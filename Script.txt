local function getMyNukes()

local function getNukesByTier()
    local tiers = {}

    local workspaceNukes = workspace:FindFirstChild("Nukes")

    
    if workspaceNukes then 
        for _, nuke in pairs(workspaceNukes:GetChildren()) do 
            table.insert(allNukes, nuke) 
        end 
    end
    for _, nuke in pairs(getMyNukes()) do 
        table.insert(allNukes, nuke)

        if nuke and nuke.Parent then

                table.insert(tiers[tier], nuke)
            end
        end
    end

    return tiersend
